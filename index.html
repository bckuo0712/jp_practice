function setupCanvas(can, strokesArr, type) { let drawing = false; let timer = null; const ctx = can.getContext('2d'); const getXY = (e) => { const rect = can.getBoundingClientRect(); // 支援觸控與滑鼠 const clientX = e.touches ? e.touches[0].clientX : e.clientX; const clientY = e.touches ? e.touches[0].clientY : e.clientY; return { x: (clientX - rect.left) * (can.width / rect.width), y: (clientY - rect.top) * (can.height / rect.height) }; }; const start = (e) => { // 如果該格已經正確，禁止再畫 if ((type === 'H' && hiraDone) || (type === 'K' && kataDone)) return; drawing = true; clearTimeout(timer); // 使用者開始畫，立即取消之前的判別計時器 const p = getXY(e); ctx.beginPath(); ctx.moveTo(p.x, p.y); // 初始化這一筆劃的座標陣列 strokesArr.push([[Math.round(p.x)], [Math.round(p.y)], []]); // 移除錯誤紅框 document.getElementById(type === 'H' ? 'boxH' : 'boxK').classList.remove('is-error'); }; const move = (e) => { if (!drawing) return; if (e.cancelable) e.preventDefault(); // 防止網頁捲動 const p = getXY(e); ctx.lineTo(p.x, p.y); ctx.stroke(); const last = strokesArr[strokesArr.length - 1]; last[0].push(Math.round(p.x)); last[1].push(Math.round(p.y)); }; const stop = () => { if (!drawing) return; drawing = false; // 設定延遲判別：如果 800ms 內沒繼續畫下一筆，就送出辨識 clearTimeout(timer); timer = setTimeout(() => { if (strokesArr.length > 0) { checkSingle(type); } }, 800); }; // 事件綁定 can.addEventListener('touchstart', start, { passive: false }); can.addEventListener('touchmove', move, { passive: false }); can.addEventListener('touchend', stop); can.addEventListener('mousedown', start); can.addEventListener('mousemove', move); // 這裡改綁在 window 避免滑鼠移出畫布放開失靈 window.addEventListener('mouseup', stop); }
